Odyssey ERP — Directory Structure
Verified: 2026-08-02

/cmd
  /odyssey            # HTTP server and CLI entrypoint
  /worker             # Asynq worker
  /bootstrap-admin    # Development/admin bootstrap command

/internal             # Application modules (modular monolith)
  /app                # Configuration, router wiring, middleware
  /accounting         # GL, reports, periods, mappings, bank reconciliation
  /ap, /ar            # Payables and receivables
  /auth, /rbac        # Authentication and authorization
  /crm, /hr, /payroll, /tax
  /inventory, /wms, /mrp, /procurement, /sales, /delivery
  /projects, /pos, /portal, /api
  /analytics, /boardpack, /consol, /close, /insights, /audit
  /finance, /fixedassets, /fx, /notifications
  /masterdata, /users, /roles, /search, /dashboard
  /integration         # Cross-module integration adapters and tests
  /jobs, /observability, /perf
  /platform            # Database, cache, and HTTP infrastructure
  /shared              # Cross-module helpers
  /sqlc                # Generated sqlc bindings
  /testing, /e2e, /view

/jobs                 # Top-level Asynq wiring and job handlers
/report               # Gotenberg client and report handler
/migrations           # Database migrations (000001 through 000075)
/sql/queries          # sqlc query definitions
/sqlc.yaml            # sqlc configuration
/web
  /static             # CSS and JavaScript
  /templates          # SSR layouts, partials, pages, and reports
/scripts              # Seed, finance, and maintenance commands
/deploy               # Monitoring and deployment assets
/samples              # Sample data and fixtures
/testing              # Shared test assets
/docs                 # Current and archived documentation
